# Dependency runtime, dipin persis (==) dan bukan rentang (>=).
#
# Alasan pin: kontrak OpenAPI di openapi.json dihasilkan oleh FastAPI,
# Pydantic, dan Starlette. Kalau versinya bebas, dua orang yang
# menjalankan scripts/export_openapi.py bisa menghasilkan kontrak yang
# berbeda, dan tipe TypeScript di frontend ikut berbeda tanpa ada
# perubahan kode sama sekali. Sudah terbukti: naik satu versi patch
# FastAPI menambah field input dan ctx di schema ValidationError.
#
# Kalau menaikkan versi salah satu paket di bawah, jalankan ulang
# scripts/export_openapi.py dan ikut commit hasilnya di PR yang sama.

fastapi==0.141.1
uvicorn[standard]==0.53.0

# Starlette sebenarnya dependency turunan FastAPI, tapi sengaja dipin
# di sini karena ikut menentukan bentuk kontrak OpenAPI.
starlette==1.6.0

sqlalchemy[asyncio]==2.0.54
asyncpg==0.31.0
psycopg[binary]==3.3.6
alembic==1.20.0

pydantic==2.13.5
pydantic-settings==2.15.0
python-dotenv==1.2.3
email-validator==2.2.0

# Verifikasi id_token OIDC (SCRUM-90). Extra cryptography wajib:
# Zitadel menandatangani dengan RS256/ES256 lewat JWKS, dan tanpa
# extra itu python-jose hanya sanggup algoritma HMAC.
python-jose[cryptography]==3.5.0

# Enkripsi kredensial produk AI (SCRUM-114). Fernet, kunci dari
# CREDENTIAL_ENCRYPTION_KEY. Dipin langsung karena sebelumnya hanya
# ikut terbawa sebagai dependency python-jose.
cryptography==44.0.0

# Klien HTTP ke Zitadel: discovery, token, JWKS, userinfo. Dipakai saat
# runtime, jadi harus di sini dan bukan di requirements-dev.txt; tanpa
# ini setiap login di image Docker gagal 500.
httpx==0.28.1

# Penghitung token lokal untuk perkiraan biaya (SCRUM-244). Encoding
# cl100k_base, tanpa panggilan ke penyedia. Dipin langsung karena dipakai
# saat ringkasan sesi, bukan hanya saat pengembangan.
tiktoken==0.12.0

# Ekspor audit log ke PDF (PBI-18 AC6). Lisensi BSD, tanpa browser atau
# dependency sistem, jadi image Docker tidak perlu paket tambahan.
reportlab==5.0.1

# Unggah berkas bukti jawaban manual (PBI-22 AC7, SCRUM-233). FastAPI butuh
# paket ini untuk membaca multipart/form-data (UploadFile, Form); tanpa ini
# aplikasi gagal start begitu route unggah terdaftar.
python-multipart==0.0.33
